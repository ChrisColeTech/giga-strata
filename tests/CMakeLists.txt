# One executable per test file (R29), registered with ctest so scripts\run-tests.bat runs them.
function(giga_strata_test name source)
    qt_add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE giga_core Qt6::Test)
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/support)
    target_compile_definitions(${name} PRIVATE GIGA_STRATA_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/core/data")
    giga_strata_warnings(${name})
    add_test(NAME ${name} COMMAND ${name})
endfunction()

# Controller tests link the app layer; fixtures are read in place and copied before any write.
function(giga_strata_app_test name source)
    giga_strata_test(${name} ${source})
    target_link_libraries(${name} PRIVATE giga_app)
    target_compile_definitions(${name} PRIVATE GIGA_STRATA_APP_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/app/fixtures")
endfunction()
